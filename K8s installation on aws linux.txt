#!/bin/bash
dnf install -y conntrack-tools iproute-tc

# 1. Disable swap (EC2 usually has it disabled by default, but verify)
swapoff -a
sed -i '/swap/d' /etc/fstab

# 2. Configure network prerequisites for bridging traffic
cat <<EOF | tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

modprobe overlay
modprobe br_netfilter

cat <<EOF | tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF

sysctl --system

# 3. Install and configure containerd
dnf update -y
dnf install -y containerd

mkdir -p /etc/containerd
containerd config default | tee /etc/containerd/config.toml >/dev/null
sed -i 's/SystemdCgroup = false/SystemdCgroup = true/g' /etc/containerd/config.toml
systemctl restart containerd
systemctl enable containerd

# 4. Set SELinux to permissive mode (required for Kubernetes)
setenforce 0
sed -i 's/^SELINUX=enforcing$/SELINUX=permissive/' /etc/selinux/config

# 5. Add the Kubernetes RPM Repository (v1.31)
cat <<EOF | tee /etc/yum.repos.d/kubernetes.repo
[kubernetes]
name=Kubernetes
baseurl=https://k8s.io
enabled=1
gpgcheck=1
gpgkey=https://k8s.iorepodata/repomd.xml.key
EOF

# 6. Clean package cache and install components
dnf clean all
dnf install -y kubelet kubeadm kubectl
systemctl enable --now kubelet
