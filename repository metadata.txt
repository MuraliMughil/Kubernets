
Step 1: Rewrite the Repository File (Removing the strict exclusion rule)
cat <<EOF > /etc/yum.repos.d/kubernetes.repo
[kubernetes]
name=Kubernetes
baseurl=https://pkgs.k8s.io/core:/stable:/v1.31/rpm/
enabled=1
gpgcheck=1
gpgkey=https://pkgs.k8s.io/core:/stable:/v1.31/rpm/repodata/repomd.xml.key
EOF



Step 2: Clean the Cache and Install Components

# Clean cached repository data
dnf clean all

# Re-run the installation
dnf install -y kubelet kubeadm kubectl

# Enable the system agent background service
systemctl enable --now kubelet


Step 3: Run the Verification Checks Again

kubeadm version
kubectl version --client
systemctl status kubelet


sudo dnf install -y conntrack-tools iproute-tc
kubectl apply -f https://raw.githubusercontent.com/projectcalico/calico/v3.30.3/manifests/calico.yaml
sudo kubeadm init --pod-network-cidr=10.244.0.0/16 --apiserver-advertise-address=<private_IP>
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config
